cmake_minimum_required(VERSION 3.14)
project(imu_web_demo CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(EHAL_IMU_SDK_DIR "" CACHE PATH "Path to the installed EHAL SDK")
if(EHAL_IMU_SDK_DIR STREQUAL "")
  message(FATAL_ERROR "Set EHAL_IMU_SDK_DIR to work_card_SDK_V1")
endif()
set(EHAL_IMU_INCLUDE_DIR "${EHAL_IMU_SDK_DIR}/include")
set(EHAL_IMU_LIBRARY "${EHAL_IMU_SDK_DIR}/lib/libehal_imu.so")
if(NOT EXISTS "${EHAL_IMU_INCLUDE_DIR}/ehal_imu.h" OR NOT EXISTS "${EHAL_IMU_LIBRARY}")
  message(FATAL_ERROR "ehal_imu header/library not found")
endif()
add_executable(imu_web_demo imu_web_demo.cpp)
target_include_directories(imu_web_demo PRIVATE "${EHAL_IMU_INCLUDE_DIR}")
target_link_libraries(imu_web_demo PRIVATE "${EHAL_IMU_LIBRARY}" pthread)
target_link_options(imu_web_demo PRIVATE -Wl,--allow-shlib-undefined)
